// Read-only: listIndexes only. Never prints the connection string or any part of it.
import { createRequire } from 'node:module';
import dns from 'node:dns';
const require = createRequire(process.argv[2]);           // repo package.json → the project's driver
const { MongoClient } = require('mongodb');
dns.setServers(['8.8.8.8', '1.1.1.1']);                   // as the project's test:integration script does

const uri = process.env.MONGODB_URI;
if (!uri) { console.log('MONGODB_URI not set in the env file'); process.exit(2); }

// Redact anything that could come from the URI before printing any message.
let host = '', user = '', pass = '';
try { const u = new URL(uri); host = u.hostname; user = decodeURIComponent(u.username); pass = decodeURIComponent(u.password); }
catch { console.log('URI could not be parsed; refusing'); process.exit(2); }
const secrets = [uri, host, user, pass].filter((s) => s && s.length > 2);
const redact = (s) => {
  let out = String(s);
  for (const x of secrets) out = out.split(x).join('[redacted]');
  return out.replace(/[A-Za-z0-9._-]*mongodb\.net(:\d+)?/g, '[redacted-host]').replace(/mongodb(\+srv)?:\/\/\S+/g, '[redacted-uri]');
};

// Same host rule as tests/integration/_harness.js: only the test cluster.
if (!host.includes('civicconnecttest')) { console.log('Host check FAILED: not the test cluster; refusing to connect'); process.exit(2); }
console.log('Host check passed (host contains the test-cluster marker; host not printed).');

const EXPECTED = {
  requestHistory: ['requestId_1_requestVersion_1', 'actorId_1_idempotencyKey_1'],
  requests: ['reference_1', 'campusId_1_requesterId_1_submittedAt_-1', 'campusId_1_status_1_submittedAt_-1',
    'campusId_1_categoryId_1_assigneeId_1_status_1', 'campusId_1_assigneeId_1_status_1',
    'campusId_1_submittedAt_1_categoryId_1', 'description_text_location_text'],
  users: ['email_1'], sessions: ['lastSeenAt_1'],
  notifications: ['userId_1_read_1_createdAt_-1', 'sourceHistoryId_1_userId_1', 'createdAt_1'],
  auditLog: ['eventId_1', 'requestId_1_occurredAt_1', 'recordedAt_1'],
  reportingCounts: ['campusId_1_categoryId_1_status_1'], categories: ['campusId_1_code_1'],
  groups: ['campusId_1_name_1'], roles: ['code_1'],
};

const client = new MongoClient(uri, { serverSelectionTimeoutMS: 20000, appName: 'readonly-index-listing' });
try {
  await client.connect();
  const db = client.db();
  console.log(`Database is "civicconnect": ${db.databaseName === 'civicconnect'}`);
  for (const [coll, expected] of Object.entries(EXPECTED)) {
    console.log(`\n== ${coll}`);
    let idx;
    try { idx = await db.collection(coll).listIndexes().toArray(); }
    catch (e) {
      console.log(`  listIndexes refused: name=${e.name} code=${e.code} codeName=${e.codeName}`);
      console.log(`  message: ${redact(e.message)}`);
      continue;
    }
    for (const i of idx) {
      const props = [i.unique && 'UNIQUE', i.partialFilterExpression && `PARTIAL ${JSON.stringify(i.partialFilterExpression)}`,
        i.expireAfterSeconds != null && `TTL ${i.expireAfterSeconds}s`, i.weights && `weights ${JSON.stringify(i.weights)}`,
        i.sparse && 'SPARSE'].filter(Boolean).join(', ');
      const tag = i.name === '_id_' ? '(default)' : expected.includes(i.name) ? '(declared)' : '(NOT DECLARED)';
      console.log(`  ${i.name}  keys ${JSON.stringify(i.key)}${props ? '  [' + props + ']' : ''}  ${tag}`);
    }
    const names = idx.map((i) => i.name);
    const missing = expected.filter((n) => !names.includes(n));
    console.log(`  declared but ABSENT: ${missing.length ? missing.join(', ') : 'none'}`);
  }
} catch (e) {
  console.log(`Connection/command failed: name=${e.name} code=${e.code} codeName=${e.codeName}`);
  console.log(`message: ${redact(e.message)}`);
} finally {
  await client.close();
}
