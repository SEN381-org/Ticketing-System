// Duplicate-refusal checks against the test cluster with the application credential.
// Usage: node --env-file=<repo>/.env --import "<dns override>" duplicate-refusal.mjs <repo root>
// Prints no URI, host, password or duplicated value.
import path from 'node:path';
import { pathToFileURL } from 'node:url';
import { randomBytes, randomInt, randomUUID } from 'node:crypto';

const repo = path.resolve(process.argv[2] ?? '.');
const load = async (rel) => import(pathToFileURL(path.join(repo, rel)).href);

const { default: mongoose } = await load('node_modules/mongoose/index.js');
mongoose.set('autoIndex', false);   // the app credential cannot build indexes; do not try
mongoose.set('autoCreate', false);

const { TEST_CLUSTER_MARKER } = await load('scripts/lib/clusterMarkers.js');
const uri = process.env.MONGODB_URI;
if (!uri) { console.log('NOT RUN: MONGODB_URI is not set'); process.exit(2); }
let host;
try { host = new URL(uri).hostname; } catch { console.log('NOT RUN: URI could not be parsed'); process.exit(2); }
if (!host.includes(TEST_CLUSTER_MARKER)) { console.log('NOT RUN: host is not the test cluster'); process.exit(2); }
console.log('Host check   : passed (host contains the recorded test marker; host not printed)');

const { default: User } = await load('src/models/User.js');
const { default: Request } = await load('src/models/Request.js');
const { default: RequestHistory } = await load('src/models/RequestHistory.js');
const { default: Category } = await load('src/models/Category.js');
const { default: Group } = await load('src/models/Group.js');

await mongoose.connect(uri, { serverSelectionTimeoutMS: 15000 });
const { authInfo } = await mongoose.connection.db.admin().command({ connectionStatus: 1 });
const users = (authInfo?.authenticatedUsers ?? []).map((u) => u.user);
console.log(`Server user  : ${users.join(', ')}`);
console.log(`Database     : ${mongoose.connection.db.databaseName}`);
if (users.length !== 1 || users[0] !== 'civicconnect-app' || mongoose.connection.db.databaseName !== 'civicconnect') {
  console.log('NOT RUN: expected the civicconnect-app credential on database civicconnect');
  await mongoose.disconnect();
  process.exit(2);
}
console.log(`Started (UTC): ${new Date().toISOString()}`);
console.log();

const oid = () => new mongoose.Types.ObjectId();
const tag = randomBytes(4).toString('hex');
const redact = (msg) => String(msg).replace(/dup key: \{.*\}/s, 'dup key: [removed]');
const indexOf = (msg) => (/index: (\S+) dup key/.exec(String(msg)) ?? [])[1] ?? '(not named)';

let failures = 0;
async function check(id, label, model, first, second, filter, expectedIndex) {
  console.log(`${id}  ${label}`);
  try {
    await model.create([first]);
    console.log('  first insert : accepted');
  } catch (e) {
    console.log(`  first insert : REFUSED code ${e.code ?? '-'}: ${redact(e.message)}`);
    failures += 1;
    return;
  }
  try {
    await model.create([second]);
    console.log('  duplicate    : ACCEPTED (expected a refusal with code 11000)');
    failures += 1;
  } catch (e) {
    const ok = e.code === 11000 && indexOf(e.message) === expectedIndex;
    console.log(`  duplicate    : refused, code ${e.code ?? '-'}${e.codeName ? ` ${e.codeName}` : ''}, index ${indexOf(e.message)}`);
    console.log(`  message      : ${redact(e.message)}`);
    if (!ok) failures += 1;
  }
  const n = await model.countDocuments(filter);
  console.log(`  documents with this key after the attempt: ${n} (expected 1)`);
  if (n !== 1) failures += 1;
  console.log();
}

const campusId = oid();
const now = new Date();
const history = (o) => ({
  campusId, changeType: 'status', fromStatus: 'Received', toStatus: 'Rejected', body: `dup-check ${tag}`,
  actorRole: 'Coordinator', occurredAt: now, ...o,
});

// L2 DEF-014: same actorId and idempotencyKey on different requests and versions.
const actor14 = oid();
const key14 = randomUUID();
await check('L2 ', 'DEF-014  requestHistory (actorId, idempotencyKey)', RequestHistory,
  history({ requestId: oid(), requestVersion: 1, actorId: actor14, idempotencyKey: key14 }),
  history({ requestId: oid(), requestVersion: 1, actorId: actor14, idempotencyKey: key14 }),
  { actorId: actor14, idempotencyKey: key14 }, 'actorId_1_idempotencyKey_1');

// L2c control: the partial filter applies only to string keys, so two null keys are both accepted.
{
  const actor = oid();
  console.log('L2c  DEF-014 control: same actorId, idempotencyKey null, different requests');
  try {
    await RequestHistory.create([history({ requestId: oid(), requestVersion: 1, actorId: actor, idempotencyKey: null })]);
    await RequestHistory.create([history({ requestId: oid(), requestVersion: 1, actorId: actor, idempotencyKey: null })]);
    const n = await RequestHistory.countDocuments({ actorId: actor });
    console.log(`  both accepted; documents for this actor: ${n} (expected 2)`);
    if (n !== 2) failures += 1;
  } catch (e) {
    console.log(`  REFUSED code ${e.code ?? '-'}: ${redact(e.message)} (expected both accepted)`);
    failures += 1;
  }
  console.log();
}

// L3 DEF-015: same requestId and requestVersion with different keys.
const request15 = oid();
await check('L3 ', 'DEF-015  requestHistory (requestId, requestVersion)', RequestHistory,
  history({ requestId: request15, requestVersion: 1, actorId: oid(), idempotencyKey: randomUUID() }),
  history({ requestId: request15, requestVersion: 1, actorId: oid(), idempotencyKey: randomUUID() }),
  { requestId: request15, requestVersion: 1 }, 'requestId_1_requestVersion_1');

// L4 DEF-017: same email.
const email = `dup-${tag}@example.test`;
const user = () => ({ campusId, email, displayName: 'Duplicate check', roles: ['Requester'], passwordHash: 'not-a-real-hash' });
await check('L4 ', 'DEF-017  users (email)', User, user(), user(), { email }, 'email_1');

// S1 FR-2.6: same request reference.
const reference = `CC-2099-${String(randomInt(1_000_000)).padStart(6, '0')}`;
const requester = oid();
const request = () => ({
  campusId, categoryId: oid(), reference, location: 'Duplicate check', description: 'Synthetic request for the duplicate-refusal checks',
  securityCategory: false, status: 'Received', requesterId: requester, submittedById: requester,
  submittedAt: now, statusChangedAt: now, version: 0,
});
await check('S1 ', 'FR-2.6   requests (reference)', Request, request(), request(), { reference }, 'reference_1');

// S2: same category code in one campus.
const code = `dup-${tag}`;
await check('S2 ', '-        categories (campusId, code)', Category,
  { campusId, code, name: 'Duplicate check' }, { campusId, code, name: 'Duplicate check 2' }, { campusId, code }, 'campusId_1_code_1');

// S3: same group name in one campus.
const name = `dup-${tag}`;
await check('S3 ', '-        groups (campusId, name)', Group,
  { campusId, name }, { campusId, name }, { campusId, name }, 'campusId_1_name_1');

console.log(`Ended (UTC)  : ${new Date().toISOString()}`);
console.log(`RESULT ${failures === 0 ? 'PASS' : 'FAIL'}: 7 checks, ${failures} not as expected`);
await mongoose.disconnect();
process.exit(failures === 0 ? 0 : 1);
